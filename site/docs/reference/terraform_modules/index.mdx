import DocCards from '@site/src/components/DocCards';
import {useCurrentSidebarCategory, useDocsVersion} from '@docusaurus/plugin-content-docs/client';

export function ModuleCards() {
  const {docs} = useDocsVersion();
  const items = useCurrentSidebarCategory().items.map((item) => {
    // The excerpt drops underscores, so cut the "The <module> module" lead-in and keep the first sentence
    const description = (docs[item.docId]?.description ?? '').replace(/^The \S+ module /, '').split(/\.\s+(?=[A-Z])/)[0];
    return {
      title: item.href.split('/').filter(Boolean).pop(),
      description: description.charAt(0).toUpperCase() + description.slice(1).replace(/\.$/, ''),
      link: item.href,
    };
  });
  return <DocCards columns={2} items={items} />;
}
